import heapq
from collections import Counter

# Take input
text = input("Enter a string: ")

# Count frequency
freq = Counter(text)

# Create min heap
heap = []
for i, (ch, f) in enumerate(freq.items()):
    heap.append((f, i, ch))

heapq.heapify(heap)

# Build Huffman Tree
count = len(heap)

while len(heap) > 1:

    f1, _, left = heapq.heappop(heap)
    f2, _, right = heapq.heappop(heap)

    heapq.heappush(heap, (f1 + f2, count, (left, right)))
    count += 1

# Generate Huffman Codes
codes = {}

def generate(node, code):

    if isinstance(node, str):
        codes[node] = code if code else "0"
        return

    left, right = node

    generate(left, code + "0")
    generate(right, code + "1")


if heap:
    generate(heap[0][2], "")

# Display codes
print("\nHuffman Codes:")
for ch, code in codes.items():
    print(ch, ":", code)

# Encode original string
encoded = ""

for ch in text:
    encoded += codes[ch]

print("\nEncoded String:", encoded)

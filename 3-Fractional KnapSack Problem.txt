# Number of items
n = int(input("Enter number of items: "))

items = []

# Take value and weight of each item
for i in range(n):
    value, weight = map(float, input("Enter value and weight: ").split())
    items.append((value, weight))

# Take knapsack capacity
capacity = float(input("Enter knapsack capacity: "))

# Sort items by value/weight ratio in descending order
items.sort(key=lambda x: x[0] / x[1], reverse=True)

profit = 0

# Select items using Greedy Strategy
for value, weight in items:

    # If complete item fits in the knapsack
    if capacity >= weight:
        profit += value          # Add complete value
        capacity -= weight       # Reduce remaining capacity

    # If complete item does not fit
    else:
        # Take only the required fraction
        profit += value * (capacity / weight)
        break                    # Knapsack is full

# Display maximum profit
print("Maximum Profit:", profit)

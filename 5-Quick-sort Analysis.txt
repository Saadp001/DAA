import random

# Quick Sort function
def quick_sort(arr, low, high, randomized=False):

    if low < high:

        # Select random pivot if randomized
        if randomized:
            p = random.randint(low, high)
            arr[p], arr[high] = arr[high], arr[p]

        # Select last element as pivot
        pivot = arr[high]

        # Partition the array
        i = low - 1

        for j in range(low, high):

            if arr[j] <= pivot:
                i += 1
                arr[i], arr[j] = arr[j], arr[i]

        # Place pivot in correct position
        arr[i + 1], arr[high] = arr[high], arr[i + 1]

        p = i + 1

        # Sort left and right subarrays
        quick_sort(arr, low, p - 1, randomized)
        quick_sort(arr, p + 1, high, randomized)


# Take input
arr = list(map(int, input("Enter array elements: ").split()))

# Create two copies
a = arr.copy()
b = arr.copy()

# Deterministic Quick Sort
quick_sort(a, 0, len(a) - 1)

# Randomized Quick Sort
quick_sort(b, 0, len(b) - 1, True)

# Display results
print("Deterministic Quick Sort:", a)
print("Randomized Quick Sort:", b)
